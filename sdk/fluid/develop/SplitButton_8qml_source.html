237 function _insideMenu(item: Item): bool {
238 // Menu can already be marked deleted while focus teardown signals arrive.
239 // Read the Item references captured at opening instead of touching that Menu.
240 const content = control._menuContentItem;
241 const container = control._menuPopupItem;
242 if (!item || (!content && !container))
243 return false;
244 let cursor = item;
245 while (cursor) {
246 if (cursor === content || cursor === container)
247 return true;
248 cursor = cursor.parent;
249 }
250 return false;
251 }

253 function _placeMenu() {
254 if (!menu || !control.Window.window)
255 return;
256 const viewport = control.Window.window.contentItem;
257 const anchor = trailing.mapToItem(viewport, 0, 0);
258 const whole = control.mapToItem(viewport, 0, 0);
259 const menuWidth = menu.width > 0 ? menu.width : menu.implicitWidth;
260 const menuHeight = menu.height > 0 ? menu.height : menu.implicitHeight;
261 const left = Math.max(0, menu.leftMargin);
262 const right = viewport.width - Math.max(0, menu.rightMargin);
263 const top = Math.max(0, menu.topMargin);
264 const bottom = viewport.height - Math.max(0, menu.bottomMargin);
265 const candidates = _mirrored
266 ? [anchor.x + trailing.width - menuWidth, anchor.x,
267 whole.x + control.width - menuWidth, whole.x]
268 : [anchor.x, anchor.x + trailing.width - menuWidth,
269 whole.x, whole.x + control.width - menuWidth];
270 let chosenX = candidates[0];
271 for (let index = 0; index < candidates.length; ++index) {
272 if (candidates[index] >= left && candidates[index] + menuWidth <= right) {
273 chosenX = candidates[index];
274 break;
275 }
276 }
277 chosenX = Math.max(left, Math.min(Math.max(left, right - menuWidth), chosenX));
278 const surfaceTop = anchor.y + layout.surfaceY;
279 const below = surfaceTop + _surfaceHeight + MD.Tokens.splitButton.menuAnchorGap;
280 const above = surfaceTop - MD.Tokens.splitButton.menuAnchorGap - menuHeight;
281 const belowSpace = bottom - below;
282 const aboveSpace = surfaceTop - MD.Tokens.splitButton.menuAnchorGap - top;
283 const preferAbove = belowSpace < menuHeight && (above >= top || aboveSpace > belowSpace);
284 const chosenY = preferAbove ? above : below;
285 menu.parent = trailing;
286 menu.x = chosenX - anchor.x;
287 menu.y = chosenY - anchor.y;
288 }

362 function onActiveFocusItemChanged() {
363 const focused = control.Window.window.activeFocusItem;
364 if (control._insideMenu(focused)) {
365 control._menuOwnedFocus = true;
366 control._outsideMenuFocus = null;
367 } else if (control._menuClosing) {
368 // Qt restores the previous focus with PopupFocusReason or OtherFocusReason.
369 // A pointer/keyboard move during dismissal belongs to the user instead.
370 const focusedControl = focused as T.Control;
371 if (focusedControl && focusedControl.focusReason !== Qt.PopupFocusReason
372 && focusedControl.focusReason !== Qt.OtherFocusReason) {
373 control._restoreMenuFocus = false;
374 control._outsideMenuFocus = focused;
375 }
376 } else if (control.expanded) {
377 const focusedControl = focused as T.Control;
378 if (focusedControl && focusedControl.focusReason !== Qt.PopupFocusReason
379 && focusedControl.focusReason !== Qt.OtherFocusReason) {
380 control._menuOwnedFocus = false;
381 control._outsideMenuFocus = focused;
382 }
383 // Popup removes focus before aboutToHide, so defer relinquishing ownership.
384 Qt.callLater(() => {
385 if (control.Window.window && control.expanded && !control._menuClosing
386 && !control._insideMenu(control.Window.window.activeFocusItem))
